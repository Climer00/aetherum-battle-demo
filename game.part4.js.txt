state.hero.hp=Math.max(0,state.hero.hp-fd.total);
state.hitFlash=12;state.shake=8;state.hero.pose="hurt";
line+=" — HIT — "+fd.text+" dmg";log(line,"hit");
if(isShove){state.shovePenalty=1;log("Shove: −1 to hit until your turn ends.","warn");}
showDiceTheater(atk,dmgText,callout);
if(state.hero.hp<=0){setTimeout(()=>{state.screen="lose";state.uiDirty=true;},900);state.uiDirty=true;return;}
}else{
line+=" — MISS";log(line,"miss");showDiceTheater(atk,"",callout);
if(isShove){state.shovePenalty=1;log("Shove still applies: −1 to hit.","warn");}
}
state.animTimer=40;state._afterEnemy=true;state.uiDirty=true;
}
function skipRiposte(){
if(state.phase!=="riposte_window")return;
state.riposteReady=false;state.animTimer=0;
log("Riposte skipped (Setup stays armed).","sys");state.uiDirty=true;afterEnemyOrRiposte();
}
function doRiposte(){
if(state.phase!=="riposte_window")return;
if(state.bank<1){log("No banked AP!","warn");return;}
state.bank-=1;state.riposteReady=false;state.phase="resolve";state.telegraph=null;
state.hero.pose="riposte";state.animTimer=0;state.windupArmed=false;state.enemyPhase=null;voice("riposte");
const usedSetup=!!state.riposteSetup;
if(usedSetup)state.riposteSetup=false;
const atk=rollAtkD20(state.hero.atkBonus+1);const nat=atk.nat;
let callout=nat===20?"CRIT!":(nat===1?"FUMBLE!":"");
let line="RIPOSTE (1 bank): "+fmtD20(atk)+" vs AC "+state.enemy.ac;let dmgText="";
if(nat===1){line+=" — FUMBLE — MISS";log(line,"miss");showDiceTheater(atk,"",callout);}
else if(atk.total>=state.enemy.ac||nat===20){
let dmg=roll(state.hero.dmg);
if(usedSetup){const ex=d(dmg.sides);dmg.rolls.push(ex);dmg.total+=ex;dmg.sum+=ex;dmg.n+=1;log("Setup: +1 dmg die = "+ex,"hit");}
if(nat===20){const ex=d(dmg.sides);dmg.rolls.push(ex);dmg.total+=ex;dmg.sum+=ex;dmg.n+=1;}
const fd=fmtDmg(dmg,0);dmgText=fd.text;
state.enemy.hp=Math.max(0,state.enemy.hp-fd.total);
state.hitFlash=14;state.shake=8;state.enemy.pose="hurt";
line+=" — HIT — "+fd.text+" dmg";if(usedSetup)line+=" (setup die)";log(line,"hit");
showDiceTheater(atk,dmgText,callout);
if(state.enemy.hp<=0){setTimeout(()=>{state.screen="win";state.uiDirty=true;},800);state.uiDirty=true;return;}
}else{line+=" — MISS";log(line,"miss");showDiceTheater(atk,"",callout);}
state.animTimer=35;state._riposteDone=true;state._afterEnemy=true;state.uiDirty=true;
}
function afterEnemyOrRiposte(){
state.enemy.pose="idle";state.hero.pose="idle";state.telegraph=null;
if(state.enemy.hp<=0||state.hero.hp<=0)return;
if(state._riposteDone){state._riposteDone=false;state._afterEnemy=false;state.windupArmed=false;startPlayerTurn(false);return;}
if(state.enemyPhase==="windup"||state.windupArmed){
state.enemyPhase="heavy";state.windupArmed=false;state.enemy.pose="heavy_blow";
log("Heavy Blow lands!","warn");enemyAttack("2d8+3","Heavy Blow",false);return;
}
if(state._afterEnemy){state._afterEnemy=false;startPlayerTurn(false);}
state.uiDirty=true;
}
