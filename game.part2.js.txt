function startPlayerTurn(first){
state.phase="player";state.selectedMod=null;if(!first)state.turn++;
state.ap=5;
if(state.guardTurns>0){state.guardTurns--;if(state.guardTurns<=0){state.hero.tempAc=0;log("Guard fades.","sys");}}
for(const id of DEFAULT_HAND){if(state.discardedOnce.has(id))continue;if(!state.inHand.has(id)){state.hand.push(id);state.inHand.add(id);}}
state.hand=DEFAULT_HAND.filter(id=>state.inHand.has(id));
if(!first)log("— Turn "+state.turn+": "+state.ap+" AP —","sys");
state.uiDirty=true;
}
function totalAp(){return state.ap+state.bank;}
function spendAp(n){
if(totalAp()<n)return false;
const fromBank=Math.min(state.bank,n);
state.bank-=fromBank;
const fromAp=n-fromBank;
state.ap-=fromAp;
log("Spent "+fromBank+" bank + "+fromAp+" AP (AP "+state.ap+"/5, bank "+state.bank+"/4).","sys");
state.uiDirty=true;return true;
}
function endPlayerTurn(){
if(state.ap>0){const room=4-state.bank;const banked=Math.min(room,state.ap);
if(banked>0){state.bank+=banked;log("Banked "+banked+" → "+state.bank+"/4","sys");}
if(state.ap>banked)log((state.ap-banked)+" AP lost (full).","warn");}
state.ap=0;state.selectedMod=null;
if(state.shovePenalty){state.shovePenalty=0;log("Shove penalty clears.","sys");}
state.uiDirty=true;beginEnemyTurn();
}
function heroAc(){return state.hero.baseAc+(state.hero.tempAc||0);}
function playCard(id){
if(state.phase!=="player")return;const c=cardById(id);
if(!c||!state.inHand.has(id))return;
if(totalAp()<c.cost){log("Need "+c.cost+" AP for "+c.name+" (have "+state.ap+" AP + "+state.bank+" bank).","warn");return;}
if(c.type==="modifier"){
if(state.selectedMod===id){state.selectedMod=null;log("Modifier cleared.","sys");}
else{state.selectedMod=id;log(c.name+" ready","sys");}
state.uiDirty=true;return;
}
if(!spendAp(c.cost))return;removeFromHand(id,c);voice(c.id);
if(c.id==="guard"){
state.hero.pose="guard";state.hero.tempAc=2;state.guardTurns=1;
log("Guard: +2 armor (AC "+heroAc()+").","sys");state.animTimer=Math.max(state.animTimer,28);
}else if(c.id==="sidestep"){
state.hero.pose="sidestep";state.sidestepArmed=true;
log("Sidestep: next attack vs you −2.","sys");state.animTimer=Math.max(state.animTimer,28);
}else if(c.id==="feint"){
state.hero.pose="feint";state.feintArmed=true;
log("Feint armed: next Strike +1 to hit.","sys");state.animTimer=Math.max(state.animTimer,28);
}else if(c.id==="riposte_setup"){
state.hero.pose="riposte_setup";state.riposteSetup=true;
log("Riposte Setup armed (+1 dmg die on Use).","warn");state.animTimer=Math.max(state.animTimer,28);
}else if(c.type==="maneuver"){resolveHeroAttack(c);}
state.uiDirty=true;
}
function removeFromHand(id,c){
state.inHand.delete(id);state.hand=state.hand.filter(x=>x!==id);
if(c&&c.encounterOnce){state.discardedOnce.add(id);log(c.name+" — ONCE (discarded).","warn");}
state.uiDirty=true;
}
