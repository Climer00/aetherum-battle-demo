if(totalAp()<cost){log("Need "+cost+" AP (strike"+(mod?" + "+mod.name:"")+") — have "+state.ap+" AP + "+state.bank+" bank.","warn");return;}
if(!spendAp(cost))return;
if(mod){removeFromHand(mod.id,mod);voice(mod.id);state.selectedMod=null;}
else voice("strike");
resolveHeroAttack(mod);
}
function resolveHeroAttack(mod){
state.phase="resolve";
if(mod&&mod.id==="lunge")state.hero.pose="lunge";
else if(mod&&mod.id==="spark")state.hero.pose="spark_edge";
else if(mod&&mod.id==="precise")state.hero.pose="precise_cut";
else if(mod&&mod.type==="maneuver")state.hero.pose="attack";
else state.hero.pose="basic_strike";
let hitB=(mod&&mod.hitBonus)||0;const dmgB=(mod&&mod.dmgBonus)||0;
if(state.feintArmed){hitB+=1;state.feintArmed=false;log("Feint consumed (+1 hit).","sys");}
if(state.shovePenalty){hitB-=1;log("Shove: −1 to hit.","warn");}
const atk=rollAtkD20(state.hero.atkBonus+hitB);const ac=state.enemy.ac;const nat=atk.nat;
const label=mod?(mod.type==="maneuver"?mod.name:"Strike + "+mod.name):"Strike";
let line=label+": "+fmtD20(atk)+" vs AC "+ac;let dmgText="";let callout="";
if(nat===20)callout="CRIT!";if(nat===1)callout="FUMBLE!";
if(nat===1){line+=" — FUMBLE — MISS";log(line,"miss");showDiceTheater(atk,"",callout);}
else if(atk.total>=ac||nat===20){
let dmg=roll(state.hero.dmg);
if(nat===20){const ex=d(dmg.sides);dmg.rolls.push(ex);dmg.total+=ex;dmg.sum+=ex;dmg.n+=1;log("CRIT! extra die = "+ex,"hit");}
const fd=fmtDmg(dmg,dmgB);dmgText=fd.text;
state.enemy.hp=Math.max(0,state.enemy.hp-fd.total);
state.hitFlash=12;state.shake=6;state.enemy.pose="hurt";
line+=" — HIT — "+fd.text+" dmg";log(line,"hit");
showDiceTheater(atk,dmgText,callout,dmg);
if(state.enemy.hp<=0){setTimeout(()=>{state.screen="win";state.uiDirty=true;},900);state.uiDirty=true;return;}
}else{line+=" — MISS";log(line,"miss");showDiceTheater(atk,"",callout);}
state.animTimer=35;state.uiDirty=true;
}
function afterResolve(){
state.enemy.pose="idle";state.hero.pose="idle";
if(state.enemy.hp<=0||state.hero.hp<=0)return;
if(state.phase==="resolve"){if(totalAp()>0){state.phase="player";log("AP/bank left: "+state.ap+" AP + "+state.bank+" bank.","sys");}else endPlayerTurn();}
state.uiDirty=true;
}
const PATTERN=["club","windup","club","windup","shove","club","windup"];
function beginEnemyTurn(){
state.phase="enemy";
const act=PATTERN[state.enemyPattern%PATTERN.length];
state.enemyPattern++;state.enemyPhase=act;
if(act==="windup"){
state.telegraph="WIND-UP!";state.enemy.pose="windup";state.windupArmed=true;
voice("windup");log("Dock Brute Wind-Up!","warn");
state.riposteReady=state.bank>=1;state.phase="riposte_window";
if(state.riposteReady){state.animTimer=0;log(">>> RIPOSTE available (1 bank) — choose Use or Skip <<<","warn");}
else{state.animTimer=90;log("(Need bank for Riposte.)","sys");}
}else if(act==="club"){state.telegraph=null;state.enemy.pose="club";voice("club");log("Dock Brute swings Club!","sys");enemyAttack("1d8+3","Club",false);}
else if(act==="heavy"){state.telegraph=null;state.enemy.pose="attack";log("Dock Brute Heavy Blow!","warn");enemyAttack("2d8+3","Heavy Blow",false);}
else if(act==="shove"){state.telegraph=null;state.enemy.pose="shove";voice("shove");log("Dock Brute Shoves!","sys");enemyAttack("1d6+2","Shove",true);}
state.uiDirty=true;
}
